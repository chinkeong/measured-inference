===== THINKING =====
Design a single-file HTML animated aquarium. Canvas 2D, full-window, no external resources, CONFIG object, lots of creatures, smooth 60fps, no console errors.

Architecture plan:

- HTML skeleton: canvas fills window, resize handler, devicePixelRatio-aware (capped at 2 for performance).
- CONFIG object at the top: counts, speeds, intervals, day/night cycle seconds.
- Utilities: rand, randRange, clamp, lerp, TAU, pick.
- Time: requestAnimationFrame with delta time, global time accumulator, speed multiplier scaling dt for creatures.

Scene layer order (back to front):
1. Water gradient background (vertical gradient, dark blue at top → teal at bottom). Precompute gradient per resize.
2. Caustic light rays — slow moving light shafts from top. Draw as rotated semi-transparent trapezoids with gradient, 'lighter' or 'screen' composite, low alpha.
3. Distant layer (depth of field): some background fish silhouettes, blurred/dim? Can add faint distant fish for depth. Keep subtle.
4. Bubbles behind some creatures? Bubbles can be behind creatures but in front of background.
5. Sea floor: sand with gentle undulation, pebbles, rocks.
6. Background seaweed (darker, behind creatures), rocks, coral.
7. Bottom-dwelling creatures: starfish, crabs, lobsters, shells.
8. Mid-ground creatures: tropical fish, angelfish, schooling fish, turtle, seahorse (near anemone/seaweed).
9. Jellyfish (upper half).
10. Foreground anemone, foreground seaweed, foreground bubbles.
11. Motes.
12. Day/night overlay: darkness tint with alpha oscillation; night glow for jellyfish.

function drawRock(g,x,y,r,alt){
  const n=8, pts=[];
  for(let i=0;i<n;i++){
    const a=i/n*TAU, rr=r*rand(0.7,1.25);
    pts.push([x+Math.cos(a)*rr*1.3, y+Math.sin(a)*rr*0.75-r*0.3]);
  }
  g.beginPath();
  g.moveTo((pts[n-1][0]+pts[0][0])/2,(pts[n-1][1]+pts[0][1])/2);
  for(let i=0;i<n;i++){
    const p=pts[i], q=pts[(i+1)%n];
    g.quadraticCurveTo(p[0],p[1],(p[0]+q[0])/2,(p[1]+q[1])/2);
  }
  g.closePath();
  const rg=g.createLinearGradient(x,y-r,x,y+r);
  rg.addColorStop(0, alt?'#5d7186':'#54677a');
  rg.addColorStop(1,'#33414f');
  g.fillStyle=rg; g.fill();
  g.strokeStyle='rgba(190,220,235,0.18)'; g.lineWidth=1.5; g.stroke();
  g.fillStyle='rgba(70,140,110,0.35)';
  for(let i=0;i<6;i++){
    g.beginPath();
    g.arc(x+rand(-r*0.7,r*0.7), y+rand(-r*0.3,r*0.5), rand(1,2.5), 0, TAU);
    g.fill();
  }
}
function drawBranchCoral(g,x,y,s,cols){
  g.lineCap='round';
  const branch=(bx,by,ang,len,d)=>{
    if(d===0) return;
    const tx=bx+Math.cos(ang)*len, ty=by+Math.sin(ang)*len;
    g.strokeStyle = d>2 ? cols[0] : cols[1];
    g.lineWidth = d*1.9;
    g.beginPath(); g.moveTo(bx,by); g.lineTo(tx,ty); g.stroke();
    const k = d>3 ? 3 : 2;
    for(let i=0;i<k;i++) branch(tx,ty, ang+rand(-0.85,0.85), len*rand(0.6,0.75), d-1);
  };
  for(let i=0;i<3;i++) branch(x+rand(-6,6), y, -Math.PI/2+rand(-0.5,0.5), s*rand(0.8,1.1), i===1?4:3);
}
function drawBrainCoral(g,x,y,s){
  g.save();
  g.beginPath(); g.ellipse(x,y,s*1.25,s*0.78,0,Math.PI,TAU); g.closePath();
  const rg=g.createLinearGradient(x,y-s,x,y);
  rg.addColorStop(0,'#e2946a'); rg.addColorStop(1,'#b05a3e');
  g.fillStyle=rg; g.fill(); g.clip();
  g.strokeStyle='rgba(120,45,30,0.55)'; g.lineWidth=2;
  for(let i=0;i<6;i++){
    const yy=y-s*0.72+i*s*0.27;
    g.beginPath();
    for(let xx=x-s*1.25; xx<=x+s*1.25; xx+=4){
      const w=yy+Math.sin(xx*0.16+i*2.1)*3.2;
      if(xx===x-s*1.25) g.moveTo(xx,w); else g.lineTo(xx,w);
    }
    g.stroke();
  }
  g.restore();
}

/* ============================== UPDATE ============================== */
function update(sdt){
  time.t += sdt;
  const t = time.t;

  if(CONFIG.dayNightSeconds>0){
    const c=(time.rt/CONFIG.dayNightSeconds)%1;
    time.dayF = 0.5+0.5*Math.cos(c*TAU);
  } else time.dayF=1;
  time.night = 1-time.dayF;

  for(const f of tropicals){
    f.turnT-=sdt;
    if(f.turnT<=0){ f.ty=rand(H*0.22,H*0.68); f.turnT=rand(3,7); }
    f.by += (f.ty-f.by)*Math.min(1,sdt*0.6);
    f.x += f.dir*f.sp*sdt;
    if(f.dir>0 && f.x>W-36) f.dir=-1;
    else if(f.dir<0 && f.x<36) f.dir=1;
    f.y = f.by + Math.sin(t*f.bobW+f.ph)*f.bobA;
    f.tail = Math.sin(t*7+f.ph);
  }
  for(const f of angels){
    f.x += f.dir*f.sp*sdt;
    if(f.dir>0 && f.x>W-60) f.dir=-1;
    else if(f.dir<0 && f.x<60) f.dir=1;
    f.y = f.by + Math.sin(t*f.bobW+f.ph)*f.bobA;
    f.fin = Math.sin(t*2.2+f.ph);
  }
  if(school) updateSchool(school,sdt);

  for(const j of jellies){
    j.pulse += sdt/1.7;
    if(j.pulse>=1){ j.pulse=0; j.vy-=rand(30,44); j.vx+=rand(-10,10); }
    j.vy += 13*sdt;
    j.vx *= Math.max(0,1-0.5*sdt);
    j.vx += Math.sin(t*0.4+j.ph)*1.5*sdt;
    const top=H*0.06, bot=H*0.5;
    if(j.y>bot) j.vy -= (j.y-bot)*1.6*sdt;
    if(j.y<top) j.vy += (top-j.y)*1.6*sdt;
    j.vx=clamp(j.vx,-30,30); j.vy=clamp(j.vy,-60,30);
    j.x+=j.vx*sdt; j.y+=j.vy*sdt;
    if(j.x<-70) j.x=W+70; else if(j.x>W+70) j.x=-70;
    j.y=clamp(j.y,-30,H*0.62);
  }

  for(const c of crabs){
    c.t-=sdt;
    if(c.st===0){
      if(c.t<=0){ c.st=1; c.t=rand(0.7,1.5); if(Math.random()<0.6) c.dir*=-1; }
    } else {
      c.x += c.dir*c.sp*(1+0.35*Math.sin(c.lp))*sdt;
      if(c.t<=0){ c.st=0; c.t=rand(1.2,3); }
    }
    c.lp += sdt*(c.st?26:2);
    c.cd -= sdt;
    if(c.cd<=0){
      c.cd=0.5;
      const check=(x,y)=>{
        const dx=c.x-x, dy=c.y-y;
        return (Math.abs(dy)<70 && Math.abs(dx)<95) ? dx : 0;
      };
      let flee=0;
      for(const f of tropicals){ flee=check(f.x,f.y); if(flee) break; }
      if(!flee) for(const f of angels){ flee=check(f.x,f.y); if(flee) break; }
      if(!flee && turtles.length){ flee=check(turtles[0].x,turtles[0].y); }
      if(flee){ c.dir=flee>0?1:-1; c.st=1; c.t=Math.max(c.t,0.9); }
    }
    c.x=clamp(c.x,26,W-26);
    if(c.x<=26||c.x>=W-26) c.dir*=-1;
    c.y=floorY(c.x)-7*c.s;
  }

  if(lobsters.length){
    const L=lobsters[0];
    L.t-=sdt;
    if(L.st==='walk'){
      L.x+=L.dir*15*sdt;
      if(L.t<=0){ L.st='pause'; L.t=rand(1,2); }
    } else if(L.st==='pause'){
      if(L.t<=0){ L.st='show'; L.t=rand(2.2,3); }
    } else if(L.t<=0){ L.st='walk'; L.t=rand(2.5,4); }
    L.raise += ((L.st==='show'?1:0)-L.raise)*Math.min(1,sdt*3);
    if(L.x<70) L.dir=1;
    if(L.x>W-70) L.dir=-1;
    L.y=floorY(L.x);
  }

  for(const s of starfishArr){
    s.rot += 0.006*sdt*s.dr;
    s.ph  += sdt*0.3;
  }
  for(const s of seahorses){
    s.x = s.ax + Math.sin(t*0.6+s.ph)*4;
    s.y = s.y0 + Math.sin(t*0.9+s.ph)*7;
  }
  for(const T of turtles){
    const a=Math.sin(t*0.13+T.ph)*0.5;
    T.x += T.dir*Math.cos(a)*26*sdt;
    if(T.x>W-80) T.dir=-1; else if(T.x<80) T.dir=1;
    T.y = T.yBase + Math.sin(t*0.17+T.ph+1.3)*H*0.085;
    T.tilt = a*0.5;
  }
  if(clown){
    const c=clown;
    c.cyc += sdt;
    const hid=(c.cyc%9)<2.4;
    c.al += ((hid?0:1)-c.al)*Math.min(1,sdt*2.5);
    c.x = anemone.x + Math.cos(t*0.55+c.ph)*30;
    c.y = anemone.y - 34 + Math.sin(t*0.85+c.ph)*10;
    c.dir = Math.sin(t*0.55+c.ph) < 0 ? 1 : -1;
  }

  updMotes(motesB,sdt); updMotes(motesF,sdt);

  time.spawn += sdt*1000;
  const iv=CONFIG.bubbleIntervalMs;
  while(time.spawn>=iv){
    time.spawn-=iv;
    if(bubbles.length<CONFIG.maxBubbles) spawnBubble();
  }
  for(let i=bubbles.length-1;i>=0;i--){
    const b=bubbles[i];
    b.y -= b.vy*sdt;
    b.x = b.x0 + Math.sin(b.y*0.02+b.ph)*b.amp;
    if(b.y<-14){ bubbles[i]=bubbles[bubbles.length-1]; bubbles.pop(); }
  }
  for(const d of distant){
    d.x += d.sp*sdt;
    if(d.x<-40) d.x=W+40; else if(d.x>W+40) d.x=-40;
    d.y += Math.sin(t*0.3+d.ph)*3*sdt;
  }
}

function updMotes(arr,dt){
  for(const m of arr){
    m.y += m.vy*dt;
    m.x += Math.sin(time.t*0.3+m.ph)*6*dt;
    if(m.y<-8){ m.y=H+8; m.x=rand(0,W); }
    if(m.x<-8) m.x=W+8; else if(m.x>W+8) m.x=-8;
  }
}
function spawnBubble(){
  const vent = Math.random()<0.65 ? pick(vents) : rand(20,W-20);
  const x0 = clamp(vent+rand(-6,6), 4, W-4);
  const r = rand(1.8,5.5);
  bubbles.push({x0:x0, x:x0, y:floorY(vent)-rand(0,14), r:r,
    vy:rand(16,30)+r*2.5, ph:rand(0,TAU), amp:rand(2.5,8),
    front:Math.random()<0.25});
}
function updateSchool(s,dt){
  s.turn += ((Math.random()-0.5)*1.4 - s.turn*0.5)*dt;
  s.heading += s.turn*dt;
  const mx=170, top=H*0.2, bot=H*0.74;
  let sx=0, sy=0;
  if(s.cx<mx) sx=(mx-s.cx)/mx; else if(s.cx>W-mx) sx=-((s.cx-(W-mx))/mx);
  if(s.cy<top+70) sy=((top+70)-s.cy)/70*0.7; else if(s.cy>bot) sy=-((s.cy-bot)/70)*0.7;
  if(sx||sy){
    const des=Math.atan2(sy,sx);
    const k=Math.min(1,Math.max(Math.abs(sx),Math.abs(sy)));
    s.heading += angDiff(des,s.heading)*Math.min(1,dt*2.5)*k;
  }
  s.cx += Math.cos(s.heading)*46*dt;
  s.cy += Math.sin(s.heading)*26*dt;
  s.cx = clamp(s.cx,120,W-120);
  s.cy = clamp(s.cy,top,bot);
  for(const m of s.members){
    m.ret-=dt;
    if(m.ret<=0){ m.tx=rand(-58,58); m.ty=rand(-42,42); m.ret=rand(0.8,2.4); }
    m.ox += (m.tx-m.ox)*Math.min(1,dt*2.5);
    m.oy += (m.ty-m.oy)*Math.min(1,dt*2.5);
  }
}

/* ============================== RENDER ============================== */
function render(){
  const t=time.t, n=time.night;
  ctx.globalCompositeOperation='source-over';
  ctx.globalAlpha=1;
  ctx.fillStyle=bgGrad; ctx.fillRect(0,0,W,H);

  // caustic shafts
  ctx.globalCompositeOperation='screen';
  const ca=lerp(1,0.4,n);
  for(const s of shafts){
    const x=s.fx*W+Math.sin(time.rt*s.spd*6+s.ph)*46;
    const a=s.a*(0.7+0.3*Math.sin(time.rt*0.4+s.ph*2))*ca;
    ctx.save(); ctx.translate(x,-20); ctx.rotate(s.rot);
    ctx.globalAlpha=a;
    ctx.drawImage(shaftSpr,-s.w/2,0,s.w,H*1.15);
    ctx.restore();
  }
  ctx.globalCompositeOperation='source-over';
  ctx.globalAlpha=1;

  ctx.fillStyle=surfGrad; ctx.fillRect(0,0,W,H*0.16);
  ctx.strokeStyle='rgba(190,235,245,0.10)'; ctx.lineWidth=1.5;
  for(let i=0;i<3;i++){
    ctx.beginPath();
    const yy=5+i*6;
    for(let x=0;x<=W;x+=24){
      const yy2=yy+Math.sin(x*0.02+time.rt*(0.6+i*0.2)+i*2)*1.8;
      if(x===0) ctx.moveTo(x,yy2); else ctx.lineTo(x,yy2);
    }
    ctx.stroke();
  }

  drawMotes(motesB,0.5);
  drawDistant();
  drawBubbles(false);
  ctx.drawImage(floorCv,0,0,W,H);
  drawSeaweed(false);

  for(const s of starfishArr) drawStar(s);
  for(const c of crabs) drawCrab(c);
  for(const L of lobsters) drawLobster(L);

  for(const T of turtles) drawTurtle(T);
  for(const f of tropicals) drawTropical(f);
  for(const f of angels) drawAngel(f);
  if(school) drawSchool();
  if(clown) drawClown(clown);
  drawAnemone();
  for(const s of seahorses) drawSeahorse(s);

  drawSeaweed(true);
  drawBubbles(true);
  drawMotes(motesF,1);

  // day/night darkening
  if(CONFIG.dayNightSeconds>0 && n>0.01){
    ctx.globalCompositeOperation='multiply';
    const k=n*0.78;
    ctx.fillStyle='rgb('+(lerp(255,92,k)|0)+','+(lerp(255,110,k)|0)+','+(lerp(255,178,k)|0)+')';
    ctx.fillRect(0,0,W,H);
  }
  // bioluminescent glow (jellyfish always faint; stronger at night; angels at night)
  ctx.globalCompositeOperation='lighter';
  const jg=0.07+n*0.4;
  for(const j of jellies){
    const s=j.r*4.8;
    ctx.globalAlpha=jg;
    ctx.drawImage(glowSpr,j.x-s/2,j.y-j.r*0.5-s/2,s,s);
  }
  const ag=n*0.16;
  if(ag>0.01){
    for(const f of angels){
      const s=70*f.s;
      ctx.globalAlpha=ag;
      ctx.drawImage(glowSpr,f.x-s/2,f.y-s/2,s,s);
    }
  }
  ctx.globalAlpha=1;
  ctx.globalCompositeOperation='source-over';

  // coin glint on the treasure chest
  const sp=Math.pow(Math.max(0,Math.sin(t*0.8)),8);
  if(sp>0.05){
    ctx.globalAlpha=sp;
    ctx.drawImage(sparkleSpr,chestPos.x-10,chestPos.y-10,20,20);
    ctx.globalAlpha=1;
  }
  ctx.drawImage(vigCv,0,0,W,H);
}

function drawMotes(arr,k){
  ctx.fillStyle='#cfe9f2';
  const t=time.t;
  for(let i=0;i<arr.length;i++){
    const m=arr[i];
    ctx.globalAlpha=(0.06+0.05*Math.sin(t*0.8+m.ph))*k;
    ctx.beginPath(); ctx.arc(m.x,m.y,m.r,0,TAU); ctx.fill();
  }
  ctx.globalAlpha=1;
}
function drawDistant(){
  ctx.fillStyle='rgba(46,84,110,0.55)';
  for(const d of distant){
    ctx.save();
    ctx.translate(d.x,d.y);
    ctx.scale(d.sp>0?1:-1,1);
    ctx.scale(d.s,d.s);
    ctx.beginPath();
    ctx.ellipse(0,0,11,5.5,0,0,TAU);
    ctx.moveTo(-9,0); ctx.lineTo(-16,-5); ctx.lineTo(-16,5); ctx.closePath();
    ctx.fill();
    ctx.restore();
  }
}
function drawBubbles(front){
  ctx.globalAlpha=0.85;
  for(const b of bubbles){
    if(b.front!==front) continue;
    const d=b.r*3.4;
    ctx.drawImage(bubbleSpr,b.x-d/2,b.y-d/2,d,d);
  }
  ctx.globalAlpha=1;
}
function drawSeaweed(front){
  const t=time.t;
  ctx.lineCap='round';
  for(const s of seaweeds){
    if(s.front!==front) continue;
    const baseX=s.x, baseY=floorY(s.x)+3;
    const segLen=s.h/s.seg;
    let px=baseX, py=baseY;
    ctx.strokeStyle=s.col;
    for(let i=0;i<s.seg;i++){
      const sway=Math.sin(t*s.spd+s.ph+i*0.55)*(2+i*1.5)*(front?1:0.8);
      const nx=baseX+sway, ny=py-segLen;
      ctx.lineWidth=Math.max(1.2, s.w-i*0.55);
      ctx.beginPath();
      ctx.moveTo(px,py);
      ctx.quadraticCurveTo(px,py-segLen*0.5,nx,ny);
      ctx.stroke();
      px=nx; py=ny;
    }
  }
}
function drawAnemone(){
  const t=time.t, cur=Math.sin(t*0.5+anemone.ph)*5;
  ctx.lineCap='round';
  const tn=anemone.tents;
  for(let i=0;i<tn.length;i++){
    const e=tn[i];
    const sway=Math.sin(t*1.2+e.ph)*7+cur;
    const len=e.len*(1+0.07*Math.sin(t*1.9+e.ph));
    const x0=anemone.x+e.ox, y0=anemone.y-4;
    const tx=x0+sway, ty=y0-len;
    ctx.strokeStyle=i%2?'#a85fc4':'#8a4fb0';
    ctx.lineWidth=3.2;
    ctx.beginPath();
    ctx.moveTo(x0,y0);
    ctx.quadraticCurveTo(x0+sway*0.4, y0-len*0.45, tx, ty);
    ctx.stroke();
    ctx.fillStyle='#e79df0';
    ctx.beginPath(); ctx.arc(tx,ty,1.8,0,TAU); ctx.fill();
  }
}

function drawJelly(j){
  const t=time.t;
  const k=j.pulse<0.3 ? 1-j.pulse/0.3 : 0;
  ctx.save();
  ctx.translate(j.x,j.y);
  ctx.strokeStyle=j.tc;
  ctx.lineWidth=1.4; ctx.lineCap='round';
  for(const tn of j.tent){
    const bx=tn.o*j.r*0.7;
    const len=j.r*tn.l;
    const sway=Math.sin(t*1.3+tn.p)*6-j.vx*0.25;
    const dx=-j.vx*0.12, dy=Math.max(0,j.vy)*0.06;
    ctx.beginPath();
    ctx.moveTo(bx,2);
    ctx.quadraticCurveTo(bx+sway*0.5+dx*0.5, len*0.55, bx+sway+dx, len+dy*4);
    ctx.stroke();
  }
  ctx.lineWidth=3;
  ctx.strokeStyle=j.toc;
  for(let i=0;i<4;i++){
    const bx=(i-1.5)*6;
    ctx.beginPath();
    ctx.moveTo(bx,2);
    ctx.quadraticCurveTo(bx+Math.sin(t*1.6+i)*3, 8, bx+Math.sin(t*1.6+i+1)*4, j.r*0.55);
    ctx.stroke();
  }
  ctx.scale(1+0.16*k, 1-0.2*k);
  const r=j.r;
  ctx.beginPath();
  ctx.moveTo(-r,2);
  ctx.bezierCurveTo(-r*1.02,-r*1.45, r*1.02,-r*1.45, r,2);
  for(let i=0;i<4;i++){
    const x1=r-((i+0.5)*(r/2)), x0=r-((i+1)*(r/2));
    ctx.quadraticCurveTo(x1,8,x0,2);
  }
  ctx.closePath();
  const rg=ctx.createRadialGradient(0,-r*0.55,r*0.15, 0,-r*0.35, r*1.5);
  rg.addColorStop(0,'rgba(255,255,255,0.55)');
  rg.addColorStop(0.5,j.toc);
  rg.addColorStop(1,'rgba(255,255,255,0.05)');
  ctx.fillStyle=rg; ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,0.4)'; ctx.lineWidth=1.2; ctx.stroke();
  ctx.fillStyle='rgba(255,255,255,0.35)';
  ctx.beginPath(); ctx.ellipse(0,-r*0.45,r*0.28,r*0.4,0,0,TAU); ctx.fill();
  ctx.restore();
}
function drawCrab(c){
  ctx.save();
  ctx.translate(c.x,c.y);
  ctx.scale(c.dir*c.s,c.s);
  const wamp=c.st===1?1:0.25;
  ctx.strokeStyle='#a34327'; ctx.lineWidth=2.2; ctx.lineCap='round';
  for(let i=0;i<3;i++){
    const sw=wamp*Math.sin(c.lp+i*2.1)*2.5;
    ctx.beginPath();
    ctx.moveTo(-7+i*1.5,-2+i);
    ctx.quadraticCurveTo(-11-i*2.5, 3+i, -13-i*3+sw, 8+i*1.6);
    ctx.stroke();
  }
  for(let i=0;i<3;i++){
    const sw=wamp*Math.sin(c.lp+1+i*2.1)*2.5;
    ctx.beginPath();
    ctx.moveTo(3+i*2,-2+i);
    ctx.quadraticCurveTo(8+i*2.5, 3+i, 11+i*3+sw, 8+i*1.6);
    ctx.stroke();
  }
  ctx.fillStyle='#cf5b3a';
  ctx.beginPath(); ctx.ellipse(0,-5,13.5,8.5,0,0,TAU); ctx.fill();
  ctx.fillStyle='rgba(255,220,190,0.25)';
  ctx.beginPath(); ctx.ellipse(-2,-8,8,4,0,0,TAU); ctx.fill();

function drawSeahorse(s){
  const t=time.t;
  const curl=0.5+0.5*Math.sin(t*0.45+s.ph);
  ctx.save();
  ctx.translate(s.x,s.y);
  ctx.scale(s.dir*s.s,s.s);
  // curled tail (spiral)
  ctx.strokeStyle=s.col; ctx.lineWidth=3.6; ctx.lineCap='round';
  const cx0=-2-curl*2, cy0=16+curl*2;
  let a0=Math.PI/2, r=6.5-curl*1.2;
  let px=cx0+Math.cos(a0)*r, py=cy0+Math.sin(a0)*r;
  ctx.beginPath(); ctx.moveTo(1.5,11);
  for(let k2=0;k2<6.5;k2++){
    a0+=0.95; r*=0.82;
    px=cx0+Math.cos(a0)*r; py=cy0+Math.sin(a0)*r;
    ctx.lineTo(px,py);
  }
  ctx.stroke();
  // body
  ctx.fillStyle=s.col;
  ctx.beginPath();
  ctx.moveTo(3,-21);
  ctx.bezierCurveTo(9,-14, 9.5,-4, 4.5,7);
  ctx.quadraticCurveTo(3,10, 1.5,11);
  ctx.quadraticCurveTo(-3,9, -3,4);
  ctx.bezierCurveTo(-2,-5, -1.5,-14, 0.5,-19);
  ctx.closePath(); ctx.fill();
  ctx.strokeStyle=s.colD; ctx.lineWidth=1.2;
  for(let i=0;i<4;i++){
    const yy=-16+i*5.2;
    ctx.beginPath(); ctx.moveTo(-2.6,yy); ctx.lineTo(6.5-i*1.2,yy); ctx.stroke();
  }
  // head + snout
  ctx.fillStyle=s.col;
  ctx.beginPath(); ctx.arc(3.5,-24,5,0,TAU); ctx.fill();
  ctx.strokeStyle=s.col; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(7,-24.5); ctx.lineTo(13.5,-23.5); ctx.stroke();
  ctx.fillStyle=s.col;
  ctx.beginPath(); ctx.arc(2,-29,1.8,0,TAU); ctx.fill();
  ctx.fillStyle='#20140c';
  ctx.beginPath(); ctx.arc(5,-24.5,1.2,0,TAU); ctx.fill();
  // flicking dorsal fin
  ctx.save();
  ctx.translate(8.5,-8);
  ctx.rotate(Math.sin(t*13+s.ph)*0.35);
  ctx.fillStyle='rgba(255,255,255,0.5)';
  ctx.beginPath(); ctx.moveTo(0,-2); ctx.quadraticCurveTo(5,0,0,2.4); ctx.closePath(); ctx.fill();
  ctx.restore();
  ctx.restore();
}
function hexPath(g,x,y,r){
  g.beginPath();
  for(let i=0;i<6;i++){
    const a=i*TAU/6+Math.PI/6;
    const px=x+Math.cos(a)*r, py=y+Math.sin(a)*r;
    if(i===0) g.moveTo(px,py); else g.lineTo(px,py);
  }
  g.closePath();
}

/* ============================== LOOP ============================== */
let last=performance.now();
function frame(now){
  const dt=Math.min(0.05,(now-last)/1000)||0;
  last=now;
  time.rt+=dt;
  update(dt*CONFIG.speedMultiplier);
  render();
  requestAnimationFrame(frame);
}
resize();
requestAnimationFrame(t=>{ last=t; frame(t); });
