/* ============================================================
   Helpers
   ============================================================ */
function rand(a, b){ return a + Math.random() * (b - a); }
function clamp(v, a, b){ return v < a ? a : (v > b ? b : v); }
function angDiff(a, b){
  let d = (a - b) % TAU;
  if (d >  Math.PI) d -= TAU;
  if (d < -Math.PI) d += TAU;
  return d;
}
function lerpC(c1, c2, t){
  return [
    Math.round(c1[0] + (c2[0]-c1[0]) * t),
    Math.round(c1[1] + (c2[1]-c1[1]) * t),
    Math.round(c1[2] + (c2[2]-c1[2]) * t)
  ];
}
function rgb(c){ return "rgb(" + c[0] + "," + c[1] + "," + c[2] + ")"; }

let time = 0, night = 0, brightness = 1;

function update(dt){
  const s = CONFIG.speed;

  if (CONFIG.dayNightCycle > 0){
    const p = time / CONFIG.dayNightCycle * TAU;
    brightness = 0.45 + 0.55 * (0.5 + 0.5 * Math.sin(p - Math.PI / 2));
    night = clamp((1 - brightness) / 0.55, 0, 1);
  } else { brightness = 1; night = 0; }

  // ---- tropical fish
  for (const f of tropicalFish){
    f.x += f.dir * f.spd * s * dt;
    f.y += Math.sin(time * 0.7 + f.dph) * f.dAmp * dt;
    if (f.x > W + 30){ f.dir = -1; f.x = W + 30; }
    if (f.x < -30){ f.dir = 1; f.x = -30; }
    f.y = clamp(f.y, 50, FLOOR - 40);
  }

  // ---- angelfish
  for (const a of angelfish){
    a.x += a.dir * a.spd * s * dt;
    a.y = a.baseY + Math.sin(time * 0.9 + a.bph) * a.bAmp;
    if (a.x > W + 40){ a.dir = -1; a.x = W + 40; }
    if (a.x < -40){ a.dir = 1; a.x = -40; }
    a.baseY += (a.y - a.baseY) * dt * 0.3;
    a.baseY = clamp(a.baseY, 110, FLOOR - 70);
  }

  // ---- school
  school.heading += (Math.sin(time * 0.25 + school.off) * 0.0035 +
                     Math.sin(time * 0.13 + school.off * 2) * 0.002) * s;
  if (school.cx < W * 0.18) school.heading += angDiff(0, school.heading) * 0.03;
  if (school.cx > W * 0.82) school.heading += angDiff(Math.PI, school.heading) * 0.03;
  if (school.cy < H * 0.16) school.heading += angDiff(school.heading * 2, school.heading) * 0.0;
  school.cx += Math.cos(school.heading) * 30 * s * dt;
  school.cy += Math.sin(school.heading) * 30 * 0.55 * s * dt;
  school.cy = clamp(school.cy, H * 0.16, FLOOR - 90);
  if (school.cy <= H * 0.16 + 1 || school.cy >= FLOOR - 91){
    school.heading += angDiff((Math.random() < 0.5 ? 0 : Math.PI), school.heading) * 0.05;
  }
  const k = 1 - Math.exp(-dt * 4);
  for (const m of school.members){
    const h = school.heading;
    const tx = school.cx + m.ox * Math.cos(h) - m.oy * Math.sin(h) + Math.sin(time * 3 + m.jp) * 4;
    const ty = school.cy + m.ox * Math.sin(h) + m.oy * Math.cos(h) + Math.cos(time * 2.6 + m.jp) * 3;
    m.x += (tx - m.x) * k;
    m.y += (ty - m.y) * k;
  }

  // ---- jellyfish
  for (const j of jellies){
    j.ph += j.pulse * s * dt;
    const contract = -Math.cos(j.ph);               // >0 while bell contracts
    const vy = clamp(contract, 0, 1) * 34 * s - 9;  // upward thrust minus slow sink
    j.y += vy * dt;
    j.x += (Math.sin(time * 0.3 + j.off) * 10 + j.drift) * s * dt;
    if (j.x < -40) j.x = W + 40;
    if (j.x > W + 40) j.x = -40;
    if (j.y > H * 0.55) j.y = H * 0.55;             // keep in upper half
    if (j.y < 30) j.y = 30;
  }

  // ---- crabs
  for (const c of crabs){
    c.timer -= dt;
    if (c.state === "pause"){
      if (c.timer <= 0){ c.state = "move"; c.timer = rand(0.5, 1.2); c.dir = Math.random() < 0.5 ? 1 : -1; }
      // flee if a fish swims close
      const flee = f => {
        const dx = f.x - c.x, dy = f.y - (FLOOR - 10);
        if (Math.abs(dx) < 80 && dy < 20 && dy > -90){ c.dir = dx > 0 ? -1 : 1; c.state = "move"; c.timer = 1.0; return true; }
        return false;
      };
      for (const f of tropicalFish) if (flee(f)) break;
      if (c.state === "pause") for (const a of angelfish) if (flee(a)) break;
    } else {
      c.x += c.dir * 42 * s * dt;
      c.leg += dt * 22;
      if (c.timer <= 0){ c.state = "pause"; c.timer = rand(1.5, 4.5); }
    }
    c.x = clamp(c.x, 20, W - 20);
  }

  // ---- lobsters
  for (const l of lobsters){
    l.timer -= dt;
    if (l.state === "walk"){
      l.x += l.dir * 9 * s * dt;
      l.leg += dt * 6;
      l.raise = Math.max(0, l.raise - dt * 2);
      if (l.timer <= 0){ l.state = "pause"; l.timer = rand(1.2, 2.5); l.raiseT = l.timer; }
    } else {
      l.raise = clamp(l.raise + dt * 2.5, 0, 1);
      if (l.timer <= 0){
        l.state = "walk"; l.timer = rand(3, 6);
        l.dir = Math.random() < 0.5 ? 1 : -1;
        if (l.x < 60) l.dir = 1; if (l.x > W - 60) l.dir = -1;
      }
    }
    l.x = clamp(l.x, 40, W - 40);
  }

  // ---- starfish (almost imperceptibly slow)
  for (const st of starfish){
    st.x += st.dir * st.spd * 0.35 * s * dt;
    st.rot += st.dir * 0.004 * s * dt;
    if (st.x < 20){ st.x = 20; st.dir = 1; }
    if (st.x > W - 20){ st.x = W - 20; st.dir = -1; }
  }

  // ---- seahorses
  for (const sh of seahorses){
    sh.y = sh.baseY + Math.sin(time * 0.7 + sh.ph) * 6;
    sh.tail = Math.sin(time * 0.9 + sh.ph) * 0.5;
  }

  // ---- sea turtle
  for (const t of turtles){
    if (t.x < 70) t.dir = 1;
    if (t.x > W - 70) t.dir = -1;
    const heading = t.dir * (0.12 + Math.sin(time * 0.18 + t.off) * 0.22);
    t.x += Math.cos(heading) * 26 * s * dt * t.dir;
    t.baseY += Math.sin(time * 0.25 + t.off) * 6 * s * dt;
    t.baseY = clamp(t.baseY, H * 0.3, FLOOR - 70);
    t.y = t.baseY;
    t.heading = heading;
    t.stroke += dt * 1.8 * s;
  }

  // ---- bubbles
  bubbleAcc += dt * 1000;
  while (bubbleAcc >= CONFIG.bubbleInterval){
    bubbleAcc -= CONFIG.bubbleInterval;
    if (bubbles.length < CONFIG.maxBubbles){
      const fromVent = Math.random() < 0.35;
      const vx = fromVent ? (Math.random() < 0.5 ? anemones[0].nx * W : anemones[1].nx * W) + rand(-6, 6)
                          : rand(10, W - 10);
      bubbles.push({ x: vx, y: FLOOR + rand(-8, 8), r: rand(2, 7), off: rand(0, TAU) });
    }
  }
  for (let i = bubbles.length - 1; i >= 0; i--){
    const b = bubbles[i];
    b.y -= (18 + b.r * 4) * s * dt;
    b.x += Math.sin(time * 2.2 + b.off + b.y * 0.02) * 12 * dt;
    if (b.y < -12) bubbles.splice(i, 1);
  }

  // ---- motes
  for (const m of motes){
    m.x += (m.vx * dt) / Math.max(W, 1);
    m.y += (m.vy * dt) / Math.max(H, 1);
    if (m.x < 0) m.x += 1; if (m.x > 1) m.x -= 1;
    if (m.y < 0) m.y += 1; if (m.y > 1) m.y -= 1;
  }

  // ---- clownfish bobbing in anemones
  for (const c of clownfish){
    c.bx = c.nx * W + Math.sin(time * 0.5 + c.ph) * 14;
    c.by = c.baseY + Math.sin(time * 0.8 + c.ph) * 8;
    if (Math.random() < dt * 0.3) c.dir = c.dir === 1 ? -1 : 1;
  }
}

function drawBackground(){
  const topDay = [10, 46, 82], topNight = [2, 8, 22];
  const botDay = [26, 112, 132], botNight = [7, 38, 58];
  const g = ctx.createLinearGradient(0, 0, 0, H);
  g.addColorStop(0, rgb(lerpC(topDay, topNight, night)));
  g.addColorStop(1, rgb(lerpC(botDay, botNight, night)));
  ctx.fillStyle = g;
  ctx.fillRect(0, 0, W, H);
}

function drawCaustics(){
  const a = 0.045 * brightness;
  for (let i = 0; i < 5; i++){
    const cx = (i + 0.5) / 5 * W + Math.sin(time * 0.1 + i * 1.7) * 80;
    const w1 = 26 + i * 8, w2 = 80 + i * 22;
    const g = ctx.createLinearGradient(0, 0, 0, H * 0.85);
    g.addColorStop(0, "rgba(190,235,255," + a.toFixed(3) + ")");
    g.addColorStop(1, "rgba(190,235,255,0)");
    ctx.fillStyle = g;
    ctx.beginPath();
    ctx.moveTo(cx - w1, -4);
    ctx.lineTo(cx + w1, -4);
    ctx.lineTo(cx + w2 + Math.sin(time * 0.07 + i) * 44, H * 0.85);
    ctx.lineTo(cx - w2 + Math.sin(time * 0.09 + i * 2) * 44, H * 0.85);
    ctx.closePath();
    ctx.fill();
  }
  // faint rippling surface
  ctx.strokeStyle = "rgba(210,240,255," + (0.06 * brightness).toFixed(3) + ")";
  ctx.lineWidth = 2;
  ctx.beginPath();
  for (let x = 0; x <= W; x += 14){
    const y = 5 + Math.sin(x * 0.03 + time * 1.4) * 2.5;
    x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
  }
  ctx.stroke();
}

function drawMotes(){
  for (const m of motes){
    ctx.fillStyle = "rgba(200,230,255," + (m.a * brightness).toFixed(3) + ")";
    ctx.beginPath();
    ctx.arc(m.x * W, m.y * H, m.s, 0, TAU);
    ctx.fill();
  }
}

function drawFloor(){
  const g = ctx.createLinearGradient(0, FLOOR - 12, 0, H);

function drawRock(r){
  const x = r.nx * W, y = FLOOR + 6, s = r.s;
  const c = lerpC([120, 122, 128], [55, 58, 66], night);
  const g = ctx.createLinearGradient(x, y - s, x, y + s * 0.3);
  g.addColorStop(0, rgb(lerpC(c, [255, 255, 255], 0.12 + r.tone * 0.1)));
  g.addColorStop(1, rgb(lerpC(c, [0, 0, 0], 0.25)));
  ctx.fillStyle = g;
  ctx.beginPath();
  ctx.moveTo(x - s, y + s * 0.25);
  ctx.bezierCurveTo(x - s * 1.05, y - s * 0.7, x - s * 0.3, y - s * (1.05 + r.tone * 0.2), x, y - s * 0.95);
  ctx.bezierCurveTo(x + s * 0.45, y - s * (1.02 - r.tone * 0.25), x + s * 1.05, y - s * 0.5, x + s, y + s * 0.25);
  ctx.closePath();
  ctx.fill();
}

function drawCoral(c){
  const x = c.nx * W, y = FLOOR + 10, s = c.s;
  ctx.strokeStyle = c.hue;
  ctx.lineCap = "round";
  const branch = (bx, by, ang, len, w, d) => {
    const sway = Math.sin(time * 0.8 + c.ph + d) * 0.06;
    const ex = bx + Math.cos(ang + sway) * len;
    const ey = by + Math.sin(ang + sway) * len;
    ctx.lineWidth = w * s;
    ctx.beginPath();
    ctx.moveTo(bx, by);
    ctx.lineTo(ex, ey);
    ctx.stroke();
    if (d < 3){
      branch(ex, ey, ang + sway + 0.45, len * 0.66, w * 0.7, d + 1);
      branch(ex, ey, ang + sway - 0.4, len * 0.6, w * 0.7, d + 1);
    }
  };
  branch(x, y, -Math.PI / 2, 26 * s, 7 * s, 0);
  // brain-coral base bump
  ctx.fillStyle = c.hue;
  ctx.globalAlpha = 0.85;
  ctx.beginPath();
  ctx.ellipse(x, y, 16 * s, 9 * s, 0, Math.PI, TAU);
  ctx.fill();
  ctx.globalAlpha = 1;
}

function drawShell(sh){
  const x = sh.nx * W, y = FLOOR + 12;
  ctx.save();
  ctx.translate(x, y);
  ctx.rotate(sh.rot);
  const s = sh.s * 14;
  if (sh.type === "conch"){
    ctx.fillStyle = "rgb(" + (222 - sh.tint * 40) + "," + (190 - sh.tint * 30) + "," + (150 - sh.tint * 20) + ")";
    ctx.beginPath();
    for (let a = 0; a < TAU * 2.2; a += 0.15){
      const r = s * 0.55 * Math.exp(-a * 0.28);
      const px = Math.cos(a) * r, py = Math.sin(a) * r * 0.8 - s * 0.2;
      a === 0 ? ctx.moveTo(px, py) : ctx.lineTo(px, py);
    }
    ctx.lineWidth = 4;
    ctx.strokeStyle = ctx.fillStyle;
    ctx.lineCap = "round";
    ctx.stroke();
    ctx.strokeStyle = "rgba(90,60,40,0.55)";
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(-s * 0.4, s * 0.45);
    ctx.quadraticCurveTo(0, s * 0.6, s * 0.35, s * 0.4);
    ctx.stroke();
  } else if (sh.type === "scallop"){
    ctx.fillStyle = "rgb(" + (235 - sh.tint * 30) + "," + (190 - sh.tint * 20) + "," + (200 - sh.tint * 20) + ")";
    ctx.beginPath();
    ctx.moveTo(0, s * 0.55);
    ctx.arc(0, s * 0.55, s * 0.85, -Math.PI * 0.88, -Math.PI * 0.12);
    ctx.closePath();
    ctx.fill();
    ctx.strokeStyle = "rgba(160,90,110,0.5)";
    ctx.lineWidth = 1.2;
    for (let i = 0; i < 7; i++){
      const a = -Math.PI * 0.88 + (i / 6) * Math.PI * 0.76;
      ctx.beginPath();
      ctx.moveTo(0, s * 0.55);
      ctx.lineTo(Math.cos(a) * s * 0.85, s * 0.55 + Math.sin(a) * s * 0.85);
      ctx.stroke();
    }
  } else { // cowrie
    ctx.fillStyle = "rgb(" + (240 - sh.tint * 20) + "," + (220 - sh.tint * 30) + "," + (195 - sh.tint * 35) + ")";
    ctx.beginPath();
    ctx.ellipse(0, 0, s * 0.55, s * 0.34, 0, 0, TAU);
    ctx.fill();
    ctx.strokeStyle = "rgba(120,80,60,0.6)";
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(-s * 0.5, s * 0.1);
    ctx.lineTo(s * 0.42, s * 0.12);
    ctx.stroke();
    ctx.fillStyle = "rgba(255,255,255,0.35)";
    ctx.beginPath();
    ctx.ellipse(-s * 0.12, -s * 0.12, s * 0.2, s * 0.08, -0.3, 0, TAU);
    ctx.fill();
  }
  ctx.restore();
}

function drawChest(){
  const x = chest.nx * W, y = FLOOR + 10, s = chest.s;
  ctx.save();
  ctx.translate(x, y);
  ctx.scale(s, s);
  // gold spill
  ctx.fillStyle = "rgba(255,200,80,0.8)";
  for (let i = 0; i < 6; i++){
    const a = -Math.PI + (i / 5) * Math.PI;
    ctx.beginPath();
    ctx.arc(Math.cos(a) * 14, -12 + Math.sin(a) * 5, 2.4, 0, TAU);
    ctx.fill();
  }
  // box
  ctx.fillStyle = "#5a3a22";
  ctx.beginPath();
  ctx.rect(-20, -16, 40, 18);

function drawSeaweed(sw){
  const x = sw.nx * W, y = FLOOR + 6;
  const segs = 9, len = sw.h / segs;
  ctx.strokeStyle = sw.hue;
  ctx.lineCap = "round";
  for (let pass = 0; pass < 2; pass++){
    ctx.lineWidth = pass === 0 ? 6 : 2.5;
    ctx.globalAlpha = pass === 0 ? 0.9 : 0.5;
    ctx.beginPath();
    let px = x, py = y;
    ctx.moveTo(px, py);
    for (let i = 1; i <= segs; i++){

function drawJelly(j){
  const scale = 1 + 0.22 * Math.sin(j.ph);
  const r = j.r * scale;
  // night glow
  if (night > 0.25){
    const gl = ctx.createRadialGradient(j.x, j.y, 2, j.x, j.y, r * 3);
    gl.addColorStop(0, "rgba(150,225,255," + (0.2 * night).toFixed(3) + ")");
    gl.addColorStop(1, "rgba(150,225,255,0)");
    ctx.fillStyle = gl;
    ctx.beginPath();
    ctx.arc(j.x, j.y, r * 3, 0, TAU);
    ctx.fill();
  }
  // tentacles
  const nT = 6;
  for (let k = 0; k < nT; k++){
    const f = (k / (nT - 1) - 0.5);
    const bx = j.x + f * r * 1.2;
    const len = j.r * (2.1 + Math.sin(k * 2.3) * 0.4);
    ctx.strokeStyle = "rgba(170,210,255,0.35)";
    ctx.lineWidth = 1.6;
    ctx.beginPath();
    ctx.moveTo(bx, j.y + r * 0.15);
    let px = bx, py = j.y + r * 0.15;
    for (let seg = 1; seg <= 5; seg++){
      const t = seg / 5;
      px = j.x + f * r * 1.1 + Math.sin(j.ph * 1.2 + k * 0.9 + seg * 0.8) * (3 + seg * 2.4);
      py = j.y + r * 0.15 + len * t;
      ctx.lineTo(px, py);
    }
    ctx.stroke();
    // trailing oral arms (center two, wider)
    if (k === 2 || k === 3){
      ctx.strokeStyle = "rgba(200,170,255,0.3)";
      ctx.lineWidth = 3;
      ctx.beginPath();
      ctx.moveTo(bx, j.y + r * 0.2);
      for (let seg = 1; seg <= 4; seg++){
        ctx.lineTo(j.x + f * r * 0.8 + Math.sin(j.ph + k + seg) * (4 + seg * 2), j.y + r * 0.2 + len * 0.85 * (seg / 4));
      }
      ctx.stroke();
    }
  }
  // bell
  const hueMix = j.tint < 0.5 ? [160, 200, 255] : [200, 160, 255];
  const g = ctx.createRadialGradient(j.x, j.y - r * 0.15, r * 0.1, j.x, j.y, r);
  g.addColorStop(0, "rgba(" + hueMix[0] + "," + hueMix[1] + "," + hueMix[2] + ",0.55)");
  g.addColorStop(0.75, "rgba(" + hueMix[0] + "," + hueMix[1] + "," + hueMix[2] + ",0.22)");
  g.addColorStop(1, "rgba(" + hueMix[0] + "," + hueMix[1] + "," + hueMix[2] + ",0.08)");
  ctx.fillStyle = g;
  ctx.beginPath();
  ctx.moveTo(j.x - r, j.y + r * 0.25);
  ctx.bezierCurveTo(j.x - r * 1.05, j.y - r * 1.0, j.x + r * 1.05, j.y - r * 1.0, j.x + r, j.y + r * 0.25);
  for (let k = 3; k >= 0; k--){
    const fx = j.x - r + (k / 3) * r * 2;
    const fx2 = j.x - r + ((k + 0.5) / 3) * r * 2;
    ctx.quadraticCurveTo(fx2, j.y + r * 0.45, fx, j.y + r * 0.25);
  }
  ctx.closePath();
  ctx.fill();
  ctx.strokeStyle = "rgba(220,240,255,0.35)";
  ctx.lineWidth = 1.2;
  ctx.stroke();
  // inner glow core
  ctx.fillStyle = "rgba(255,255,255," + (0.1 + 0.12 * night).toFixed(3) + ")";
  ctx.beginPath();
  ctx.ellipse(j.x, j.y - r * 0.3, r * 0.35, r * 0.28, 0, 0, TAU);
  ctx.fill();
}

function drawBubble(b){
  const g = ctx.createRadialGradient(b.x - b.r * 0.35, b.y - b.r * 0.35, b.r * 0.1, b.x, b.y, b.r);
  g.addColorStop(0, "rgba(255,255,255,0.55)");
  g.addColorStop(0.65, "rgba(180,225,255,0.1)");
  g.addColorStop(1, "rgba(190,230,255,0.3)");
  ctx.fillStyle = g;
  ctx.beginPath();
  ctx.arc(b.x, b.y, b.r, 0, TAU);
  ctx.fill();
  ctx.strokeStyle = "rgba(230,245,255,0.35)";
  ctx.lineWidth = 0.8;
  ctx.beginPath();
  ctx.arc(b.x, b.y, b.r, 0, TAU);
  ctx.stroke();
  // specular highlight
  ctx.fillStyle = "rgba(255,255,255,0.7)";
  ctx.beginPath();
  ctx.arc(b.x - b.r * 0.35, b.y - b.r * 0.4, b.r * 0.18, 0, TAU);
  ctx.fill();
}

function drawClownfish(c){
  ctx.save();
  ctx.translate(c.bx, c.by);
  ctx.scale(c.dir, 1);
  const s = c.size;
  ctx.fillStyle = "#ff7b2a";
  ctx.beginPath();
  ctx.ellipse(0, 0, s, s * 0.55, 0, 0, TAU);
  ctx.fill();
  ctx.fillStyle = "rgba(255,255,255,0.95)";
  ctx.strokeStyle = "rgba(40,20,10,0.5)";
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.rect(-s * 0.25, -s * 0.55, s * 0.3, s * 1.1);
  ctx.fill(); ctx.stroke();
  ctx.beginPath();
  ctx.rect(s * 0.35, -s * 0.45, s * 0.25, s * 0.9);
  ctx.fill(); ctx.stroke();
  ctx.fillStyle = "rgba(255,120,50,0.85)";
  ctx.beginPath();
  ctx.moveTo(-s * 0.85, 0);
  ctx.lineTo(-s * 1.4, -s * 0.45);
  ctx.lineTo(-s * 1.3, s * 0.45);
  ctx.closePath();
  ctx.fill();
  ctx.fillStyle = "#fff";
  ctx.beginPath();
  ctx.arc(s * 0.55, -s * 0.08, s * 0.14, 0, TAU);
  ctx.fill();
  ctx.fillStyle = "#1a1a1a";
  ctx.beginPath();
  ctx.arc(s * 0.6, -s * 0.08, s * 0.08, 0, TAU);
  ctx.fill();
  ctx.restore();
}

   Main loop

let last = performance.now();

function frame(now){
  let dt = (now - last) / 1000;
  last = now;
  if (dt > 0.05) dt = 0.05;
  time += dt;

  update(dt);

  drawBackground();
  drawCaustics();
  drawMotes();
  drawFloor();
  for (const r of rocks) drawRock(r);
  drawChest();
  for (const c of corals) drawCoral(c);
  for (const sh of shells) drawShell(sh);
  for (const sw of seaweeds) drawSeaweed(sw);
  for (const an of anemones) drawAnemone(an);
  for (const cl of clownfish) drawClownfish(cl);
  for (const st of starfish) drawStarfish(st);
  for (const c of crabs) drawCrab(c);
  for (const l of lobsters) drawLobster(l);
  for (const sh of seahorses) drawSeahorse(sh);
  for (const t of turtles) drawTurtle(t);
  for (const f of tropicalFish) drawTropical(f);
  for (const a of angelfish) drawAngelfish(a);
  for (const m of school.members) drawSchoolMember(m);
  for (const j of jellies) drawJelly(j);
  for (const b of bubbles) drawBubble(b);

  // gentle vignette
  const vg = ctx.createRadialGradient(W / 2, H / 2, Math.min(W, H) * 0.4, W / 2, H / 2, Math.max(W, H) * 0.75);
  vg.addColorStop(0, "rgba(0,0,10,0)");
  vg.addColorStop(1, "rgba(0,0,10," + (0.25 + 0.2 * night).toFixed(3) + ")");
  ctx.fillStyle = vg;
  ctx.fillRect(0, 0, W, H);

  requestAnimationFrame(frame);
}
