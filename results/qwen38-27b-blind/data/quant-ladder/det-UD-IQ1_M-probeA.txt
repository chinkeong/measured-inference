class MinHeap {
  constructor() { this.a = []; }
  get size() { return this.a.length; }
  push(node, pri) {
    this.a.push({ node, pri });
    this.siftUp(this.a.length - 1);
  }
  pop() {
    const top = this.a[0];
    const last = this.a.pop();
    if (this.a.length > 0) {
      this.a[0] = last;
      this.siftDown(0);
    }
    return top;
  }
  peek() { return this.a[0]; }
  siftUp(i) {
    while (i > 0) {
      const parent = Math.floor((i - 1) / 2);
      if (this.a[i].pri < this.a[parent].pri) {
        [this.a[i], this.a[parent]] = [this.a[parent], this.a[i]];
        i = parent;
      } else break;
    }
  }
  siftDown(i) {
    while (i < this.a.length) {
      const child = Math.min(i * 2 + 1, i * 2 + 2);
      const childIdx = i * 2 + 1;
      const child2 = i * 2 + 2;
      const minIdx = childIdx < this.a.length ? childIdx : child2;
      const minVal = this.a[minIdx].pri;
      if (this.a[i].pri >= minVal) {
        [this.a[i], this.a[minIdx]] = [this.a[minIdx], this.a[i]];
        i = minIdx;
      } else break;
    }
  }
}

class Graph {
  constructor() { this.nodes = new Map(); this.edges = new Map(); }
  addNode(id) { this.nodes.set(id, id); }
  addEdge(u, v, w) {
    if (!this.nodes.has(u)) this.addNode(u);
    if (!this.nodes.has(v)) this.addNode(v);
    if (!this.edges.has(u)) this.edges.set(u, []);
    this.edges.get(u).push({ to: v, weight: w });
  }
  neighbors(u) { return this.edges.get(u) || []; }
  nodeCount() { return this.nodes.size; }
}
