class MinHeap {
  constructor() { this.a = []; }
  get size() { return this.a.length; }
  push(node, pri) {
    this.a.push({ node: node, pri: pri });
    this.siftUp(this.a.length - 1);
  }
  pop(node) {
    return this.a.shift();
  }
  peek() {
    return this.a[0];
  }
  siftUp(i) {
    let j = i;
    while (j > 0) {
      let k = j - 1;
      if (this.a[k].pri < this.a[j].pri) {
        this.a[k] = this.a[j];
        j = k;
      }
      break;
    }
  }
  siftDown(i) {
    let j = i;
    while (j > 0) {
      let k = j - 1;
      if (this.a[k].pri < this.a[j].pri) {
        this.a[k] = this.a[j];
        j = k;
      }
      break;
    }
  }
}
