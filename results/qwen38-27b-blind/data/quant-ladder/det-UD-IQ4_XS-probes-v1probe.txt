function dijkstra(graph, source) {
  const dist = new Map();
  const prev = new Map();
  const visited = new Set();
  const heap = new MinHeap();

}

function reconstructPath(prev, source, target) {
  if (!prev.has(target) || prev.get(target) === null && target !== source) {
    return null;
  }
  const path = [];
  let cur = target;
  while (cur !== null) {
    path.unshift(cur);
    if (cur === source) break;
    cur = prev.get(cur);
  }
  if (path[0] !== source) return null;
  return path;
}
