class Graph {
  constructor() { this.nodes = new Set(); this.adj = new Map(); }
  addNode(id) {
    if (!this.nodes.has(id)) { this.nodes.add(id); this.adj.set(id, []); }
  }
  addEdge(u, v, w) {
    this.addNode(u); this.addNode(v);
    this.adj.get(u).push([v, w]);
  }
  neighbors(u) { return this.adj.get(u) || []; }
  get nodeCount() { return this.nodes.size; }
}

function dijkstra(graph, source) {
  const dist = new Map();
  const prev = new Map();
  const visited = new Set();
  const heap = new MinHeap();
  for (const n of graph.nodes) dist.set(n, Infinity);
  dist.set(source, 0);
  heap.push(source, 0);
  while (heap.size > 0) {
    const [u, d] = heap.pop();
    if (visited.has(u)) continue;
    visited.add(u);
    for (const [v, w] of graph.neighbors(u)) {
      const alt = d + w;
      if (alt < (dist.get(v) ?? Infinity)) {
        dist.set(v, alt);
        prev.set(v, u);
        heap.push(v, alt);
      }
    }
  }
  return { dist, prev };
}

function reconstructPath(prev, target) {
  const path = [];
  let cur = target;
  while (cur !== undefined) {
    path.push(cur);
    cur = prev.get(cur);
  }
  return path.reverse();
}
