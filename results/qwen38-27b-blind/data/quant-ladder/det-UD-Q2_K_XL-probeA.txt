this.a.push({ node, pri });
  this.siftUp(this.a.length - 1);
}
  pop() {
    if (this.size === 0) return null;
    const top = this.a[0];
    const last = this.a.pop();
    if (this.size > 0) {
      this.a[0] = last;
      this.siftDown(0);
    }
    return top;
  }
  peek() { return this.size ? this.a[0] : null; }
  siftUp(i) {
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.a[i].pri < this.a[p].pri) {
        [this.a[i], this.a[p]] = [this.a[p], this.a[i]];
        i = p;
      } else break;
    }
  }
  siftDown(i) {
    const n = this.size;
    while (true) {
      const l = 2 * i + 1, r = 2 * i + 2;
      let m = i;
      if (l < n && this.a[l].pri < this.a[m].pri) m = l;
      if (r < n && this.a[r].pri < this.a[m].pri) m = r;
      if (m === i) break;
      [this.a[i], this.a[m]] = [this.a[m], this.a[i]];
      i = m;
    }
  }
}

class Graph {
  constructor() { this.adj = new Map(); }
  addNode(id) { if (!this.adj.has(id)) this.adj.set(id, []); }
  addEdge(u, v, w) {
    this.addNode(u); this.addNode(v);
    this.adj.get(u).push({ to: v, w });
  }
  neighbors(id) { return this.adj.get(id) || []; }
  nodeCount() { return this.adj.size; }
}

function dijkstra(graph, source) {
  const dist = new Map();
  const prev = new Map();
  const heap = new MinHeap();
  for (const id of graph.adj.keys()) dist.set(id, Infinity);
  dist.set(source, 0);
  heap.push(source, 0);
  while (heap.size) {
    const { node: u, pri: du } = heap.pop();
    if (du > dist.get(u)) continue;
    for (const { to: v, w } of graph.neighbors(u)) {
      const nd = du + w;
      if (nd < dist.get(v)) {
        dist.set(v, nd);
        prev.set(v, u);
        heap.push(v, nd);
      }
    }
  }
  return { dist, prev };
}

function reconstructPath(prev, target) {
  const path = [];
  let cur = target;
  while (cur !== undefined) {
    path.unshift(cur);
    cur = prev.get(cur);
  }
  return path;
}
