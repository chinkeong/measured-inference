  schoolHeading += dt * 0.15 * spd;
  // Gently steer center toward middle of canvas
  const targetX = W * 0.5 + Math.cos(globalTime * 0.1) * W * 0.2;
  const targetY = H * 0.4 + Math.sin(globalTime * 0.13) * H * 0.15;
  schoolCenterX += (targetX / W - schoolCenterX) * dt * 0.2;
  schoolCenterY += (targetY / H - schoolCenterY) * dt * 0.2;
  schoolCenterX = clamp(schoolCenterX, 0.2, 0.8);
  schoolCenterY = clamp(schoolCenterY, 0.25, 0.6);
  for (const f of school) {
    f.phase += dt * f.offsetSpeed;
  }
}

function drawSchool() {
  const cx = schoolCenterX * W;
  const cy = schoolCenterY * H;
  for (const f of school) {
    const px = cx + f.ox * Math.cos(f.phase * 0.5) + Math.cos(schoolHeading) * f.ox;
    const py = cy + f.oy * Math.sin(f.phase * 0.5) + Math.sin(schoolHeading) * f.oy;
    const sx = clamp(px, 10, W - 10);
    const sy = clamp(py, H * 0.15, H * 0.7);

    ctx.translate(sx, sy);

// --- JELLYFISH ---
function createJelly() {
  return {
    x: rand(W * 0.15, W * 0.85),
    y: rand(H * 0.1, H * 0.45),
    size: rand(25, 45),
    pulsePhase: rand(0, TAU),
    pulseSpeed: rand(0.8, 1.4),
    vx: rand(-0.3, 0.3),
    vy: 0,
    hue: rand(270, 330),
    tentacles: 6 + Math.floor(rand(0, 4)),
    tentLen: rand(30, 55)
  };
}
const jellies = [];
for (let i = 0; i < CONFIG.jellyfishCount; i++) jellies.push(createJelly());

function updateJelly(j, dt) {
  const spd = CONFIG.speedMultiplier;
  j.pulsePhase += dt * j.pulseSpeed * TAU;
  const pulse = Math.sin(j.pulsePhase);
  // Propulsion: push up during contraction (pulse > 0)
  j.vy += pulse > 0 ? -pulse * 0.8 * spd * dt : 0;
  // Gravity/drift down
  j.vy += 0.3 * spd * dt;
  // Damping
  j.vy *= (1 - 0.5 * dt);
  j.vx += rand(-0.1, 0.1) * spd * dt;
  j.vx *= (1 - 0.3 * dt);
  j.x += j.vx * 60 * dt;
  j.y += j.vy * 60 * dt;
  // Keep in upper half
  if (j.y < H * 0.05) j.y = H * 0.05;
  if (j.y > H * 0.55) j.y = H * 0.55;
  if (j.x < j.size) j.x = j.size;
  if (j.x > W - j.size) j.x = W - j.size;
}

function drawJelly(j) {
  const pulse = Math.sin(j.pulsePhase);
  const bellScale = 1 + pulse * 0.15;
  const s = j.size * bellScale;
  ctx.save();
  ctx.translate(j.x, j.y);
  // Glow at night
  const nightGlow = CONFIG.dayNightCycle > 0 ? Math.max(0, -Math.cos(dayNightPhase)) * 0.3 : 0;
  if (nightGlow > 0.05) {
    ctx.beginPath();
    ctx.arc(0, 0, s * 1.5, 0, TAU);
    ctx.fillStyle = `hsla(${j.hue},80%,70%,${nightGlow * 0.15})`;
    ctx.fill();
  }
  // Tentacles
  const tCount = j.tentacles;
  for (let i = 0; i < tCount; i++) {
    const tx = (i / (tCount - 1) - 0.5) * s * 1.2;
    const sway = Math.sin(globalTime * 2 + i * 0.8 + j.pulsePhase * 0.5) * 8;
    ctx.beginPath();
    ctx.moveTo(tx * 0.5, s * 0.3);
    ctx.bezierCurveTo(
      tx * 0.5 + sway * 0.5, s * 0.3 + j.tentLen * 0.3,
      tx * 0.3 + sway, s * 0.3 + j.tentLen * 0.7,
      tx * 0.2 + sway * 1.2, s * 0.3 + j.tentLen
    );
    ctx.strokeStyle = `hsla(${j.hue},60%,75%,${0.3 + nightGlow * 0.3})`;
    ctx.lineWidth = 1.5;
    ctx.stroke();
  }
  // Bell
  ctx.beginPath();
  ctx.moveTo(-s * 0.7, s * 0.2);
  ctx.quadraticCurveTo(-s * 0.8, -s * 0.5, -s * 0.4, -s * 0.8);
  ctx.quadraticCurveTo(0, -s * 1.1, s * 0.4, -s * 0.8);
  ctx.quadraticCurveTo(s * 0.8, -s * 0.5, s * 0.7, s * 0.2);
  ctx.quadraticCurveTo(s * 0.4, s * 0.45, 0, s * 0.35);
  ctx.quadraticCurveTo(-s * 0.4, s * 0.45, -s * 0.7, s * 0.2);
  ctx.closePath();
  const jGrad = ctx.createRadialGradient(0, -s * 0.2, 0, 0, 0, s);
  jGrad.addColorStop(0, `hsla(${j.hue},70%,80%,${0.5 + nightGlow * 0.2})`);
  jGrad.addColorStop(1, `hsla(${j.hue},50%,50%,${0.2 + nightGlow * 0.1})`);
  ctx.fillStyle = jGrad;
  ctx.fill();
  // Inner glow
  ctx.beginPath();
  ctx.ellipse(0, -s * 0.2, s * 0.3, s * 0.4, 0, 0, TAU);
  ctx.fillStyle = `hsla(${j.hue},80%,85%,${0.15 + nightGlow * 0.2})`;
  ctx.fill();
  ctx.restore();
}

function createCrab() {
  return {
    x: rand(W * 0.1, W * 0.9),
    size: rand(18, 28),
    dir: Math.random() < 0.5 ? 1 : -1,
    state: 'idle', // idle, scuttle
    stateTimer: rand(0, 2),
    speed: rand(1, 2)
  };
}
const crabs = [];
for (let i = 0; i < CONFIG.crabCount; i++) crabs.push(createCrab());

function updateCrab(c, dt) {
  const spd = CONFIG.speedMultiplier;
  c.stateTimer -= dt * spd;
  if (c.stateTimer <= 0) {
    if (c.state === 'idle') {
      c.state = 'scuttle';
      c.dir = Math.random() < 0.5 ? 1 : -1;
      c.stateTimer = rand(0.5, 1.5);
    } else {
      c.state = 'idle';
      c.stateTimer = rand(1, 3);
    }
  }
  if (c.state === 'scuttle') {
    c.x += c.dir * c.speed * 60 * dt * spd;
  }
  c.x = clamp(c.x, c.size, W - c.size);
}

function drawCrab(c) {
  const y = floorY - c.size * 0.3;
  ctx.save();
  ctx.translate(c.x, y);
  const s = c.size;
  const legAnim = c.state === 'scuttle' ? Math.sin(globalTime * 20) * 3 : 0;
  // Legs
  for (let i = -1; i <= 1; i += 2) {
    for (let j = 0; j < 3; j++) {
      ctx.beginPath();
      ctx.moveTo(i * s * 0.3, s * 0.1);
      ctx.lineTo(i * (s * 0.5 + j * s * 0.15), s * 0.4 + legAnim * (j % 2 ? 1 : -1));
      ctx.strokeStyle = '#b35';
      ctx.lineWidth = 2;
      ctx.stroke();
    }
  }
  // Body
  ctx.beginPath();
  ctx.ellipse(0, 0, s * 0.6, s * 0.4, 0, 0, TAU);
  ctx.fillStyle = '#cc4433';
  ctx.fill();
  // Claws
  for (let i = -1; i <= 1; i += 2) {
    const clawX = i * s * 0.7;
    const clawOpen = Math.sin(globalTime * 3 + i) * 0.2 + 0.3;
    ctx.beginPath();
    ctx.ellipse(clawX, s * 0.1, s * 0.25, s * 0.15, i * 0.3, 0, TAU);
    ctx.fillStyle = '#bb3322';
    ctx.fill();
    // Pincer
    ctx.beginPath();
    ctx.moveTo(clawX + i * s * 0.2, s * 0.05);
    ctx.lineTo(clawX + i * s * 0.35, -s * 0.1 + clawOpen * s * 0.1);
    ctx.moveTo(clawX + i * s * 0.2, s * 0.15);
    ctx.lineTo(clawX + i * s * 0.35, s * 0.25 - clawOpen * s * 0.1);
    ctx.strokeStyle = '#aa2211';
    ctx.lineWidth = 2.5;
    ctx.stroke();
  }
  // Eyestalks
  for (let i = -1; i <= 1; i += 2) {
    ctx.beginPath();
    ctx.moveTo(i * s * 0.2, -s * 0.3);
    ctx.lineTo(i * s * 0.25, -s * 0.5);
    ctx.strokeStyle = '#cc4433';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.beginPath();
    ctx.arc(i * s * 0.25, -s * 0.52, s * 0.08, 0, TAU);
    ctx.fillStyle = '#fff';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(i * s * 0.25, -s * 0.52, s * 0.04, 0, TAU);
    ctx.fillStyle = '#111';
    ctx.fill();
  }
  ctx.restore();
}

    state: 'walk',
    stateTimer: rand(2, 5),
    clawRaise: 0,
    phase: rand(0, TAU)
  };
}
const lobsters = [];
for (let i = 0; i < CONFIG.lobsterCount; i++) lobsters.push(createLobster());

function updateLobster(l, dt) {
  const spd = CONFIG.speedMultiplier;
  l.phase += dt;
  l.stateTimer -= dt * spd;
  if (l.stateTimer <= 0) {
    if (l.state === 'walk') {
      l.state = 'pause';
      l.clawRaise = 1;
      l.stateTimer = rand(1, 3);
    } else {
      l.state = 'walk';
      l.dir = Math.random() < 0.7 ? l.dir : -l.dir;
      l.stateTimer = rand(3, 6);
      l.clawRaise = 0;
    }
  }
  if (l.state === 'walk') {
    l.x += l.dir * 0.5 * 60 * dt * spd;
  }
  l.clawRaise = lerp(l.clawRaise, l.state === 'pause' ? 1 : 0, dt * 3);
  l.x = clamp(l.x, l.size, W - l.size);
}

function drawLobster(l) {
  const y = floorY - l.size * 0.2;
  ctx.save();
  ctx.translate(l.x, y);
  ctx.scale(l.dir, 1);
  const s = l.size;
  const segCount = 5;
  // Tail segments
  for (let i = 0; i < segCount; i++) {
    const sx = -s * 0.5 - i * s * 0.18;
    const sw = Math.sin(l.phase * 2 + i * 0.5) * 2;
    ctx.beginPath();
    ctx.ellipse(sx, sw, s * 0.15, s * 0.1 - i * 0.01, 0, 0, TAU);
    ctx.fillStyle = `hsl(${15 + i * 3},70%,${35 + i * 5}%)`;
    ctx.fill();
  }
  // Tail fan
  const fanX = -s * 0.5 - segCount * s * 0.18;
  const fanSwing = Math.sin(l.phase * 2) * 3;
  ctx.beginPath();
  ctx.moveTo(fanX, fanSwing);
  ctx.lineTo(fanX - s * 0.2, fanSwing - s * 0.15);
  ctx.lineTo(fanX - s * 0.15, fanSwing + s * 0.05);
  ctx.lineTo(fanX - s * 0.2, fanSwing + s * 0.15);
  ctx.closePath();
  ctx.fillStyle = '#8b2020';
  ctx.fill();
  // Body
  ctx.beginPath();
  ctx.ellipse(0, 0, s * 0.4, s * 0.2, 0, 0, TAU);
  ctx.fillStyle = '#a03020';
  ctx.fill();
  // Head
  ctx.beginPath();
  ctx.ellipse(s * 0.35, 0, s * 0.15, s * 0.12, 0, 0, TAU);
  ctx.fillStyle = '#b04030';
  ctx.fill();
  // Antennae
  for (let i = -1; i <= 1; i += 2) {
    ctx.beginPath();
    ctx.moveTo(s * 0.45, i * s * 0.03);
    ctx.quadraticCurveTo(s * 0.8, i * s * 0.1, s * 1.1, i * s * 0.05 + Math.sin(l.phase * 3 + i) * 3);
    ctx.strokeStyle = '#c05040';
    ctx.lineWidth = 1.5;
    ctx.stroke();
  }
  // Claws
  const clawLift = l.clawRaise * s * 0.3;
  for (let i = -1; i <= 1; i += 2) {
    const cx = s * 0.5;
    const cy = i * s * 0.12 - clawLift;
    ctx.beginPath();
    ctx.ellipse(cx, cy, s * 0.2, s * 0.12, i * 0.2, 0, TAU);
    ctx.fillStyle = '#902818';
    ctx.fill();
    // Pincer opening
    const open = 0.3 + l.clawRaise * 0.4;
    ctx.beginPath();
    ctx.moveTo(cx + s * 0.15, cy - open * s * 0.08);
    ctx.lineTo(cx + s * 0.3, cy - open * s * 0.15);
    ctx.moveTo(cx + s * 0.15, cy + open * s * 0.08);
    ctx.lineTo(cx + s * 0.3, cy + open * s * 0.15);
    ctx.strokeStyle = '#701808';
    ctx.lineWidth = 3;
    ctx.stroke();
  }
  // Legs (small)
  for (let i = 0; i < 4; i++) {
    const lx = -s * 0.1 + i * s * 0.12;
    ctx.beginPath();
    ctx.moveTo(lx, s * 0.15);
    ctx.lineTo(lx + s * 0.05, s * 0.3);
    ctx.strokeStyle = '#803020';
    ctx.lineWidth = 2;
    ctx.stroke();
  }
  // Eyes
  for (let i = -1; i <= 1; i += 2) {
    ctx.beginPath();
    ctx.arc(s * 0.4, i * s * 0.05, s * 0.03, 0, TAU);
    ctx.fillStyle = '#111';
    ctx.fill();
  }
  ctx.restore();
}

// --- SEA TURTLE ---
function createTurtle() {
  return {
    x: rand(W * 0.2, W * 0.8),
    y: rand(H * 0.3, H * 0.6),
    angle: rand(0, TAU),
    turnRate: 0,
    speed: 0.4,
    size: rand(35, 50),
    flipperPhase: rand(0, TAU),
    arcTimer: rand(3, 7)
  };
}
const turtles = [];
for (let i = 0; i < CONFIG.turtleCount; i++) turtles.push(createTurtle());

function updateTurtle(t, dt) {
  const spd = CONFIG.speedMultiplier;
  t.flipperPhase += dt * 3 * spd;
  t.arcTimer -= dt;
  if (t.arcTimer <= 0) {
    t.turnRate = rand(-0.3, 0.3);
    t.arcTimer = rand(4, 8);
  }
  t.angle += t.turnRate * dt * spd;
  t.x += Math.cos(t.angle) * t.speed * 60 * dt * spd;
  t.y += Math.sin(t.angle) * t.speed * 40 * dt * spd;
  // Soft boundaries
  if (t.x < t.size) { t.x = t.size; t.angle = Math.PI - t.angle; }
  if (t.x > W - t.size) { t.x = W - t.size; t.angle = Math.PI - t.angle; }
  if (t.y < H * 0.15) { t.y = H * 0.15; t.angle = -t.angle; }
  if (t.y > H * 0.7) { t.y = H * 0.7; t.angle = -t.angle; }
}

function drawTurtle(t) {
  ctx.save();
  ctx.translate(t.x, t.y);
  ctx.rotate(t.angle);
  const s = t.size;
  const flap = Math.sin(t.flipperPhase) * 0.3;
  // Flippers
  for (let i = -1; i <= 1; i += 2) {
    // Front
    ctx.save();
    ctx.rotate(i * (0.5 + flap));
    ctx.beginPath();
    ctx.ellipse(s * 0.4, i * s * 0.4, s * 0.35, s * 0.1, i * 0.3, 0, TAU);
    ctx.fillStyle = '#3a6b4a';
    ctx.fill();
    ctx.restore();
    // Rear
    ctx.save();
    ctx.rotate(i * (0.4 - flap * 0.7));
    ctx.beginPath();
    ctx.ellipse(-s * 0.5, i * s * 0.3, s * 0.25, s * 0.08, i * 0.2, 0, TAU);
    ctx.fillStyle = '#3a6b4a';
    ctx.fill();
    ctx.restore();
  }
  // Shell
  ctx.beginPath();
  ctx.ellipse(0, 0, s * 0.55, s * 0.4, 0, 0, TAU);
  const shellGrad = ctx.createRadialGradient(0, 0, 0, 0, 0, s * 0.55);
  shellGrad.addColorStop(0, '#5a8a5a');
  shellGrad.addColorStop(1, '#2a5a3a');
  ctx.fillStyle = shellGrad;
  ctx.fill();
  // Scute pattern
  ctx.strokeStyle = 'rgba(20,60,30,0.6)';
  ctx.lineWidth = 1.5;
  // Center hexagon
  ctx.beginPath();
  for (let i = 0; i < 6; i++) {
    const a = (i / 6) * TAU;
    const px = Math.cos(a) * s * 0.2;
    const py = Math.sin(a) * s * 0.15;
    if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
  }
  ctx.closePath();
  ctx.stroke();
  // Outer ring
  ctx.beginPath();
  for (let i = 0; i < 8; i++) {
    const a = (i / 8) * TAU + 0.2;
    const px = Math.cos(a) * s * 0.45;
    const py = Math.sin(a) * s * 0.33;
    if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
  }
  ctx.closePath();
  ctx.stroke();
  // Lines from center to outer
  for (let i = 0; i < 8; i++) {
    const a = (i / 8) * TAU + 0.2;
    ctx.beginPath();
    ctx.moveTo(Math.cos(a) * s * 0.2, Math.sin(a) * s * 0.15);
    ctx.lineTo(Math.cos(a) * s * 0.45, Math.sin(a) * s * 0.33);
    ctx.stroke();
  }
  // Head
  ctx.beginPath();
  ctx.ellipse(s * 0.6, 0, s * 0.18, s * 0.12, 0, 0, TAU);
  ctx.fillStyle = '#4a7a4a';
  ctx.fill();
  // Eye
  ctx.beginPath();
  ctx.arc(s * 0.65, -s * 0.04, s * 0.03, 0, TAU);
  ctx.fillStyle = '#111';
  ctx.fill();
  ctx.restore();
}

  ctx.fill();
  // Tentacles
  for (let i = 0; i < a.tentacles; i++) {
    const t = i / a.tentacles;
    const angle = (t - 0.5) * 1.2;
    const sway = Math.sin(globalTime * 1.5 + a.phase + i * 0.7) * 6;
    const tx = Math.sin(angle) * a.size * 0.4 + sway;
    const ty = -a.size * (0.8 + 0.3 * Math.sin(i * 2.1));
    ctx.beginPath();
    ctx.moveTo(Math.sin(angle) * a.size * 0.2, -a.size * 0.1);
    ctx.quadraticCurveTo(tx * 0.5, ty * 0.5, tx, ty);
    ctx.strokeStyle = `hsla(${a.hue},60%,${55 + i * 2}%,0.7)`;
    ctx.lineWidth = 3;
    ctx.lineCap = 'round';
    ctx.stroke();
    // Tip
    ctx.beginPath();
    ctx.arc(tx, ty, 2.5, 0, TAU);
    ctx.fillStyle = `hsl(${a.hue},70%,70%)`;
    ctx.fill();
  }
  ctx.restore();
}

// --- BACKGROUND ---
function drawBackground() {
  const nightFactor = CONFIG.dayNightCycle > 0 ? (0.5 - 0.5 * Math.cos(dayNightPhase)) : 0;
  // Water gradient
  const grad = ctx.createLinearGradient(0, 0, 0, H);
  const topR = lerp(5, 2, nightFactor);
  const topG = lerp(20, 10, nightFactor);
  const topB = lerp(50, 25, nightFactor);
  const botR = lerp(10, 5, nightFactor);
  const botG = lerp(80, 35, nightFactor);
  const botB = lerp(120, 60, nightFactor);
  grad.addColorStop(0, `rgb(${topR},${topG},${topB})`);
  grad.addColorStop(0.6, `rgb(${lerp(10,5,nightFactor)},${lerp(50,20,nightFactor)},${lerp(80,40,nightFactor)})`);
  grad.addColorStop(1, `rgb(${botR},${botG},${botB})`);
  ctx.fillStyle = grad;
  ctx.fillRect(0, 0, W, H);

  // Caustic light rays
  ctx.save();
  ctx.globalAlpha = 0.03 + (1 - nightFactor) * 0.02;
  for (let i = 0; i < CONFIG.causticCount; i++) {
    const cx = W * (0.1 + i * 0.2) + Math.sin(globalTime * 0.2 + i * 1.5) * 40;
    const width = 30 + Math.sin(globalTime * 0.3 + i) * 15;
    ctx.beginPath();
    ctx.moveTo(cx - width * 0.3, 0);
    ctx.lineTo(cx + width * 0.3, 0);
    ctx.lineTo(cx + width, H * 0.7);
    ctx.lineTo(cx - width, H * 0.7);
    ctx.closePath();
    ctx.fillStyle = `hsl(180,30%,80%)`;
    ctx.fill();
  }
  ctx.restore();

  // Particle motes
  ctx.save();
  for (const m of motes) {
    const mx = (m.x * W + Math.sin(globalTime * m.speed * 10 + m.phase) * 20 + globalTime * m.speed * 30) % W;
    const my = (m.y * H + Math.cos(globalTime * m.speed * 5 + m.phase) * 15) % H;
    ctx.beginPath();
    ctx.arc(mx, my, m.size, 0, TAU);
    ctx.fillStyle = `rgba(180,220,255,${0.15 + (1 - nightFactor) * 0.1})`;
    ctx.fill();
  }
  ctx.restore();
}

function drawFloor() {
  const nightFactor = CONFIG.dayNightCycle > 0 ? (0.5 - 0.5 * Math.cos(dayNightPhase)) : 0;
  // Sandy floor with undulation
  ctx.beginPath();
  ctx.moveTo(0, H);
  ctx.lineTo(0, floorY + 10);
  for (let x = 0; x <= W; x += 20) {
    const y = floorY + Math.sin(x * 0.01 + 1) * 5 + Math.sin(x * 0.03) * 3;
    ctx.lineTo(x, y);
  }
  ctx.lineTo(W, H);
  ctx.closePath();
  const sandGrad = ctx.createLinearGradient(0, floorY, 0, H);
  sandGrad.addColorStop(0, `hsl(${lerp(35,25,nightFactor)},${lerp(40,25,nightFactor)}%,${lerp(55,30,nightFactor)}%)`);
  sandGrad.addColorStop(1, `hsl(${lerp(30,20,nightFactor)},${lerp(35,20,nightFactor)}%,${lerp(40,20,nightFactor)}%)`);
  ctx.fillStyle = sandGrad;
  ctx.fill();

  // Pebbles
  ctx.fillStyle = `hsla(30,20%,${lerp(45,25,nightFactor)}%,0.5)`;
  for (let i = 0; i < 20; i++) {
    const px = (i * 137.5 + 50) % W;
    const py = floorY + 8 + (i * 73.3) % 15;
    ctx.beginPath();
    ctx.ellipse(px, py, 3 + (i % 3), 2 + (i % 2), 0, 0, TAU);
    ctx.fill();
  }
}

// --- BUBBLES ---
function spawnBubble() {
  if (bubbles.length >= CONFIG.maxBubbles) return;
  bubbles.push({
    x: rand(W * 0.05, W * 0.95),
    y: floorY + rand(-5, 5),
    size: rand(3, 10),
    speed: rand(30, 70),
    wobblePhase: rand(0, TAU),
    wobbleAmp: rand(5, 15)
  });
}

function updateBubbles(dt) {
  const now = performance.now();
  if (now - lastBubbleSpawn > CONFIG.bubbleSpawnInterval) {
    spawnBubble();
    lastBubbleSpawn = now;
  }
  for (let i = bubbles.length - 1; i >= 0; i--) {
    const b = bubbles[i];
    b.y -= b.speed * CONFIG.speedMultiplier * dt;
    b.wobblePhase += dt * 3;
    b.x += Math.sin(b.wobblePhase) * b.wobbleAmp * dt;
    if (b.y < -b.size) bubbles.splice(i, 1);
  }
}

function drawBubbles() {
  for (const b of bubbles) {
    ctx.beginPath();
    ctx.arc(b.x, b.y, b.size, 0, TAU);
    ctx.fillStyle = 'rgba(180,220,255,0.15)';
    ctx.fill();
    ctx.strokeStyle = 'rgba(200,230,255,0.3)';
    ctx.lineWidth = 1;
    ctx.stroke();
    // Specular highlight
    ctx.beginPath();
    ctx.arc(b.x - b.size * 0.3, b.y - b.size * 0.3, b.size * 0.25, 0, TAU);
    ctx.fillStyle = 'rgba(255,255,255,0.4)';
    ctx.fill();
  }
}

// --- MAIN LOOP ---
function update(dt) {
  globalTime += dt;
  if (CONFIG.dayNightCycle > 0) {
    dayNightPhase += (dt / CONFIG.dayNightCycle) * TAU;
  }
  for (const f of tropicals) updateTropical(f, dt);
  for (const f of angels) updateAngel(f, dt);
  updateSchool(dt);
  for (const j of jellies) updateJelly(j, dt);
  for (const c of crabs) updateCrab(c, dt);
  for (const l of lobsters) updateLobster(l, dt);
  for (const s of starfish) updateStarfish(s, dt);
  for (const sh of seahorses) updateSeahorse(sh, dt);
  for (const t of turtles) updateTurtle(t, dt);
  updateBubbles(dt);
}

function draw() {
  drawBackground();
  drawFloor();
  // Decor (behind creatures)
  for (const sw of seaweeds) drawSeaweed(sw);
  for (const r of rocks) drawRock(r);
  for (const c of corals) drawCoral(c);
  for (const a of anemones) drawAnemone(a);
  for (const sh of shells) drawShell(sh);

  // Creatures (back to front roughly)
  for (const s of starfish) drawStarfish(s);
  for (const sh of seahorses) drawSeahorse(sh);
  for (const t of turtles) drawTurtle(t);
  for (const f of tropicals) drawTropical(f);
  for (const f of angels) drawAngel(f);
  drawSchool();
  for (const l of lobsters) drawLobster(l);
  for (const c of crabs) drawCrab(c);
  for (const j of jellies) drawJelly(j);

  drawBubbles();

  // Night overlay (subtle)
  if (CONFIG.dayNightCycle > 0) {
    const nightFactor = (0.5 - 0.5 * Math.cos(dayNightPhase));
    if (nightFactor > 0.1) {
      ctx.fillStyle = `rgba(0,0,20,${nightFactor * 0.15})`;
      ctx.fillRect(0, 0, W, H);
    }
  }
}

function loop(timestamp) {
  if (!lastTime) lastTime = timestamp;
  const dt = Math.min((timestamp - lastTime) / 1000, 0.05); // cap delta
  lastTime = timestamp;
  update(dt);
  draw();
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
